! SSH depuis le VLAN 99 uniquement
ip domain name site-c.local
username admin privilege 15 secret A_REMPLACER
crypto key generate rsa modulus 2048
ip ssh version 2
ip access-list standard SSH_VLAN99_ONLY
 permit 172.16.72.0 0.0.0.255
 deny any log
exit
line vty 0 15
 login local
 transport input ssh
 access-class SSH_VLAN99_ONLY in
exit
end
