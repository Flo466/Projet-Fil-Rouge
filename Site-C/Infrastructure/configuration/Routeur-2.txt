! LECTURE DES ACL
! Premiere regle correspondante appliquee, de haut en bas ; sinon refus implicite.
! permit = autoriser ; deny = bloquer ; ip = tous les protocoles IPv4.
! host = une IP ; any = toute adresse ; eq = port TCP/UDP.
! Wildcards : 0.0.0.31 = /27 ; 0.0.0.63 = /26 ; 0.0.0.255 = /24.
! established = bit TCP ACK ou RST, pas un veritable suivi de connexion.
! Les commentaires ! servent a la lecture du fichier, ils ne sont pas des regles.
! Une permission dans une ACL ne garantit pas le passage des autres ACL du trajet.

! ROUTEUR 2 - DMZ REVERSE PROXY
! Gi0/0/1 vers switch access VLAN30 ; Gi0/0/0 vers carte 2 Proxmox2.
! Pas de trunk, pas de sous-interface, pas de NAT entre LAN et DMZ.
enable
configure terminal
hostname R2-SITE-C
no ip domain-lookup
enable secret A_REMPLACER
interface GigabitEthernet0/0/1
 description VERS_SW_L3_ACCESS_VLAN30
 ip address 172.16.2.70 255.255.255.224
 no shutdown
exit
interface GigabitEthernet0/0/0
 description DMZ_VERS_PROXMOX2_CARTE2
 ip address 172.16.3.161 255.255.255.224
 no shutdown
exit
ip route 0.0.0.0 0.0.0.0 172.16.2.65

! FILTRE DMZ : le proxy peut joindre les deux Web et repondre aux clients/admin.
! Le switch filtre les VLAN clients. R2 filtre les paquets venant du proxy.
! Sans ce filtre, R2 donnerait acces aux autres serveurs du VLAN30.
ip access-list extended DMZ_VERS_LAN
 ! Autorise proxy -> Web1 : TCP port 80.
 permit tcp host 172.16.3.162 host 172.16.2.69 eq 80
 ! Autorise proxy -> Web1 : TCP port 443.
 permit tcp host 172.16.3.162 host 172.16.2.69 eq 443
 ! Autorise proxy -> Web2 : TCP port 80.
 permit tcp host 172.16.3.162 host 172.16.2.71 eq 80
 ! Autorise proxy -> Web2 : TCP port 443.
 permit tcp host 172.16.3.162 host 172.16.2.71 eq 443
 ! Autorise proxy -> LAN : port source 80, TCP ; retour TCP (ACK/RST).
 permit tcp host 172.16.3.162 eq 80 172.16.2.0 0.0.0.255 established
 ! Autorise proxy -> LAN : port source 443, TCP ; retour TCP (ACK/RST).
 permit tcp host 172.16.3.162 eq 443 172.16.2.0 0.0.0.255 established
 ! Autorise proxy -> VLAN99 : port source 22, TCP ; retour TCP (ACK/RST).
 permit tcp host 172.16.3.162 eq 22 172.16.2.224 0.0.0.31 established
 ! Autorise proxy -> VLAN99 : ICMP ; reponse au ping.
 permit icmp host 172.16.3.162 172.16.2.224 0.0.0.31 echo-reply
 ! Bloque tout le reste et journalise.
 deny ip any any log
exit
interface GigabitEthernet0/0/0
 ! Applique DMZ_VERS_LAN aux paquets entrant depuis ce reseau dans l'equipement.
 ip access-group DMZ_VERS_LAN in
exit

ip domain name yai.local
username admin privilege 15 secret A_REMPLACER
crypto key generate rsa modulus 2048
ip ssh version 2
ip access-list standard SSH_VLAN99
 ! Autorise les sources du VLAN 99 : 172.16.2.224/27, wildcard 0.0.0.31.
 permit 172.16.2.224 0.0.0.31
 ! Refuse les sources restantes (tout autre reseau) ; journalise les correspondances.
 deny any log
exit
line vty 0 15
 login local
 transport input ssh
 ! Filtre les sources des connexions distantes VTY avec cette ACL ; transport input ssh impose SSH.
 access-class SSH_VLAN99 in
 exec-timeout 10 0
exit
end
