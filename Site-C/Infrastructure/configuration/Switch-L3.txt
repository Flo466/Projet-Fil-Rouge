! SITE C - SWITCH L3
! Base : configuration fournie pour le switch réel.
! LAN : 172.16.2.0/24 decoupe en huit sous-reseaux /27.
! DMZ : 172.16.3.128/26 decoupee en VLAN 70 cameras et VLAN 71 reverse proxy.

enable
configure terminal
hostname SW-L3-C
no ip domain-lookup
ip routing

! =========================
! CREATION DES VLAN
! =========================
vlan 10
 name SERVICE1
vlan 20
 name SERVICE2
vlan 30
 name SERVEURS_PROXMOX2_WEB
vlan 40
 name WIFI_EMPLOYES
vlan 50
 name VOIP
vlan 60
 name WIFI_INVITES
vlan 70
 name CAMERAS_DMZ
vlan 71
 name DMZ_REVERSE_PROXY
vlan 80
 name RESERVE
vlan 99
 name MANAGEMENT
exit

! =========================
! PASSERELLES / SVI LAN
! =========================
interface Vlan10
 description GW_SERVICE1
 ip address 172.16.2.1 255.255.255.224
 ip helper-address 172.16.2.66

interface Vlan20

interface Vlan50
 description GW_VOIP

 ip helper-address 172.16.2.66
 no shutdown
interface Vlan60
 description GW_WIFI_INVITES

 ip helper-address 172.16.2.66

interface Vlan99

! Les passerelles des VLAN 70 et 71 sont sur Routeur 2.
! Le switch transporte ces VLAN mais ne porte pas leurs SVI.

! =========================
! PORTS D'ACCES - MAPPING FOURNI
! =========================

! =========================
! TRUNKS - PORTS A CONFIRMER
! =========================
! Le trunk Routeur 2 doit avoir VLAN 30 natif et VLAN 70/71 autorises.
! Le trunk Proxmox 2 doit transporter VLAN 30, 70 et 71.
! Remplacer les deux marqueurs par les vrais ports Gi1/0/X avant application.
! interface GigabitEthernet1/0/<PORT_R2>
!  description TRUNK_ROUTEUR2_PORT_A_CONFIRMER
!  switchport trunk encapsulation dot1q
!  switchport mode trunk
!  switchport trunk native vlan 30
!  switchport trunk allowed vlan 30,70,71
!  switchport nonegotiate
!  no shutdown
! exit
! interface GigabitEthernet1/0/<PORT_PROXMOX2>
!  description TRUNK_PROXMOX2_PORT_A_CONFIRMER
!  switchport trunk encapsulation dot1q
!  switchport mode trunk
!  switchport trunk native vlan 30
!  switchport trunk allowed vlan 30,70,71
!  switchport nonegotiate
!  no shutdown
! exit

! =========================
! LIEN ROUTEUR 1 : NE PAS MODIFIER
! =========================
interface GigabitEthernet1/0/24
 description TRANSIT_ROUTEUR1_EXISTANT_NE_PAS_MODIFIER
 no switchport
 ip address 172.16.137.2 255.255.255.252
 no shutdown
exit
ip route 0.0.0.0 0.0.0.0 172.16.137.1

! Routes vers les deux sous-reseaux DMZ via Routeur 2 sur VLAN 30.
ip route 172.16.3.128 255.255.255.224 172.16.2.70
ip route 172.16.3.160 255.255.255.224 172.16.2.70

! =========================
! SSH
! =========================
ip domain name yai.local
username admin privilege 15 secret A_REMPLACER
crypto key generate rsa modulus 2048
ip ssh version 2
ip access-list standard SSH_VLAN99
 permit 172.16.2.224 0.0.0.31
 deny any log
exit
line vty 0 15
 login local
 transport input ssh
 access-class SSH_VLAN99 in
 exec-timeout 10 0
exit

! =========================
